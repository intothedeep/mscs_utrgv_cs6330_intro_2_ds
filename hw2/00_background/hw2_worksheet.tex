% Letters restart at (a) for each data set.
\newcounter{qn}
\newcommand{\group}[1]{\def\grp{#1}\setcounter{qn}{0}}

% A fixed half-page box, so two land on each page.
\newcommand{\question}[1]{%
  \stepcounter{qn}%
  \noindent\begin{minipage}[t][0.48\textheight][t]{\linewidth}
    {\small \grp}\par\smallskip
    \textbf{(\alph{qn}) #1}
  \end{minipage}\par}

% Same box with the code's figure under the question: #1 file, #2 caption, #3 question.
\newcommand{\figquestion}[3]{%
  \stepcounter{qn}%
  \noindent\begin{minipage}[t][0.48\textheight][t]{\linewidth}
    {\small \grp}\par\smallskip
    \textbf{(\alph{qn}) #3}\par\medskip
    {\centering
     \includegraphics[width=\linewidth,height=0.28\textheight,keepaspectratio]{\figdir/#1.png}\par
     \captionof{figure}{#2}\par}
  \end{minipage}\par}
