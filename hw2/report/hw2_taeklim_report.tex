\prompt{\textbf{Airport data:}}

\begin{enumerate}
  \item \prompt{What is the $\alpha$ if the distribution is power law?}
  \answer{$\alpha = 1.612$, with $x_{\min} = 1$ route.
    Maximum-likelihood estimate $\hat\alpha = 1 + n / \sum_i \ln(x_i / x_{\min})$,
    $x_{\min} = \min_i x_i$, $n = 3409$.
    \begin{align*}
      &n = 3409 && \text{number of airports}\\
      &x_{\min} = \min_i x_i = 1 && \text{smallest route count}\\
      &S = \textstyle\sum_{i=1}^{3409} \ln(x_i / 1) = 5569.43 && \text{sum of log ratios}\\
      &n / S = 3409 / 5569.43 = 0.6121 && \\
      &\hat\alpha = 1 + 0.6121 = 1.612 && \text{estimate}
    \end{align*}}

  \item \prompt{What is the $\lambda$ if the distribution is exponential?}
  \answer{$\lambda = 0.0504$. MLE $\hat\lambda = 1/\bar x = 1/19.845$.
    \begin{align*}
      &\textstyle\sum_i x_i = 67652 && \text{total routes}\\
      &\bar x = 67652 / 3409 = 19.845 && \text{mean}\\
      &\hat\lambda = 1 / 19.845 = 0.0504 && \text{estimate}
    \end{align*}}

  \item \prompt{What is $[a, b]$ if the distribution is uniform?}
  \answer{$[a, b] = [1, 915]$, the minimum and maximum route counts (MLE).
    \begin{align*}
      &\hat a = \min_i x_i = 1 && \text{fewest routes}\\
      &\hat b = \max_i x_i = 915 && \text{most routes}
    \end{align*}}

  \item \prompt{What is $\mu$ and $\sigma$ if the distribution is normal?}
  \answer{$\mu = 19.845$, $\sigma = 53.506$ (sample mean and sample standard deviation
    with $n-1$).
    \begin{align*}
      &\hat\mu = \bar x = 19.845 && \text{mean, as in the exponential fit}\\
      &\textstyle\sum_i (x_i - \bar x)^2 = 9{,}756{,}906.22 && \text{sum of squared deviations}\\
      &9{,}756{,}906.22 / (3409 - 1) = 2862.94 && \text{sample variance}\\
      &\hat\sigma = \sqrt{2862.94} = 53.506 && \text{standard deviation}
    \end{align*}}

  \prompt{(These can be CDF, PDF, or QQplot)}

  For (e) to (i):
  \begin{itemize}
    \item \textbf{Samples.} $n = 3409$ random values drawn from each fitted model (seed 6330).
    \item \textbf{Colours.} Data in blue, model sample in orange.
    \item \textbf{$D$.} Two-sample Kolmogorov--Smirnov distance: the largest vertical gap
          between the two CDFs (0 = identical).
  \end{itemize}

  \item \prompt{Screen shot of the visualization of distribution of data}
  \answerfig{p1a_0_data}{Airport routes: empirical distribution ($n = 3409$). Top row on linear axes, bottom row on log-log axes.}
  \panels{%
    \textbf{PDF, linear (top left).} One tall bar at the left edge: 82\% of airports fall in the first bin (fewer than 20 routes), and the bars beyond 200 are almost flat on the floor.\par
    \textbf{CDF, linear (top right).} The curve rises almost vertically and reaches 0.92 at 60 routes (dashed line), then stays flat up to the maximum of 915.\par
    \textbf{PDF, log-log (bottom left).} Share of airports with exactly $k$ routes. The points fall along a roughly straight line from $k = 1$ (21\%) downward. In the tail many values belong to a single airport, so the points flatten into a floor at $1/3409$.\par
    \textbf{CCDF, log-log (bottom right).} Share of airports with at least $x$ routes. It is close to a straight line up to about 100 routes, then bends down and ends at the busiest airport (915 routes).\par
    \textbf{Summary.} The median airport has 4 routes and 38\% have only 1 or 2, while the busiest has 915. The mean (19.8) is five times the median and the skewness is 6.0, so a few hubs hold most of the routes.\par}

  \item \prompt{Screen shot of the visualization of distribution of data if it was power law}
  \answerfig{p1a_1_powerlaw}{Airport routes vs.\ power law ($\alpha = 1.612$, $x_{\min} = 1$). $D = 0.209$.}
  \panels{%
    \textbf{PDF (top left).} The two lines follow the same straight slope up to about 300 routes. Above that the data drops and ends near 1000, while the model continues past $10^5$ (its largest sampled airport has 291{,}593 routes).\par
    \textbf{CDF (top right).} On linear axes the two curves almost overlap. The difference is hidden in the 1.6\% of the sample that lies beyond 1000 routes.\par
    \textbf{CCDF (bottom left).} Both follow the same straight line on log-log axes up to about 100 routes. Above that the data bends down and stops at 915; the model stays straight.\par
    \textbf{QQ plot (bottom right).} The points lie on $y = x$ up to about 60 routes, then run off to the right, below the line: at the 99.5th percentile the model has 5{,}234 routes and the data 358.\par}

  \item \prompt{Screen shot of the visualization of distribution of data if it was exponential}
  \answerfig{p1a_2_exponential}{Airport routes vs.\ exponential ($\lambda = 0.0504$). $D = 0.379$.}
  \panels{%
    \textbf{PDF (top left).} The model is almost flat for small airports and then falls off sharply after about 50 routes. The data is the opposite: steep at the start and a long, slow tail. 5\% of the sample lies below 1 route, which no airport has.\par
    \textbf{CDF (top right).} The model reaches 1 by about 160 routes (its largest sampled airport has 159), while the data keeps rising slowly up to 915.\par
    \textbf{CCDF (bottom left).} The model bends down early and ends near 160 routes; the data goes on to 915. With $\lambda = 0.05$ an airport with 500 routes has probability about $e^{-25}$, yet the data has five such airports.\par
    \textbf{QQ plot (bottom right).} The points are too steep: below about 40 routes they lie below $y = x$ (the model gives small airports too many routes), and above it they rise above the line (hubs get too few).\par}

  \item \prompt{Screen shot of the visualization of distribution of data if it was uniform}
  \answerfig{p1a_3_uniform}{Airport routes vs.\ uniform on $[1, 915]$. $D = 0.857$.}
  \panels{%
    \textbf{PDF (top left).} The model is a flat line near $10^{-3}$ across the whole range; the data falls by five orders of magnitude.\par
    \textbf{CDF (top right).} The model is a straight diagonal from 1 to 915; the data reaches 0.92 by 60 routes. The largest gap between the two, $D = 0.857$, is at 60 routes.\par
    \textbf{CCDF (bottom left).} The model stays near 1 until the right edge and then drops vertically at 915.\par
    \textbf{QQ plot (bottom right).} Every point lies far below $y = x$, to the right. A typical sampled airport has about 460 routes, while the real median is 4 and 99\% of real airports have fewer than 285.\par}

  \item \prompt{Screen shot of the visualization of distribution of data if it was normal}
  \answerfig{p1a_4_normal}{Airport routes vs.\ normal ($\mu = 19.8$, $\sigma = 53.5$). $D = 0.358$.}
  \panels{%
    \textbf{PDF (top left).} The model is flat over small airports and then drops steeply after about 100 routes. 35\% of its sample is below zero, which is impossible for a route count; the log axes cannot show that part.\par
    \textbf{CDF (top right).} The model starts below zero (negative route counts) and is S-shaped around the mean; it reaches 1 near 230 routes while the data goes to 915.\par
    \textbf{CCDF (bottom left).} Below about 100 routes the model sits above the data, and its tail ends near 230 routes, far short of 915.\par
    \textbf{QQ plot (bottom right).} The points are too steep, as for the exponential: small airports get too many routes and hubs too few. The tail is thin because a normal decays like $e^{-x^2}$.\par}

  \item \prompt{What is the distribution of data based on your observation? Discuss}
  \answer{%
    \begin{center}
    \begin{tabular}{lcccc}
      \toprule
      & power law & exponential & uniform & normal \\
      \midrule
      KS distance $D$ (data vs model sample) & \textbf{0.209} & 0.379 & 0.857 & 0.358 \\
      \bottomrule
    \end{tabular}
    \end{center}

    \textbf{The airport data follows a power law, with a cutoff in the far tail.}
    It is the only one of the four models whose CCDF is a straight line on log-log
    axes. The data's CCDF is close to straight for more than two decades (1 to about
    100 routes), and the power-law sample matches it there. The other three models fail
    for structural reasons, not because a parameter is slightly off:
    \begin{itemize}
      \item \textbf{Uniform} says every route count from 1 to 915 is equally likely,
            but most airports are tiny.
      \item \textbf{Normal} is symmetric and allows negative counts, while the data is
            bounded below at 1 and skewed to the right (skewness 6.0).
      \item \textbf{Exponential} has the right shape (many small, few large), but its
            tail falls off too fast. With $\lambda = 0.05$, an airport with 500 routes
            has probability about $e^{-25}$, yet the data has five airports with more
            than 500.
    \end{itemize}
    The power law is not a perfect fit. $\hat\alpha = 1.61$ is below 2, so the fitted
    model has an infinite mean and its samples contain airports with $10^5$ routes. Real
    airports cannot grow that large: there are only 3409 airports to fly to, and
    runways, gates and slots limit the busiest hubs. That is why the data's CCDF bends
    down above about 100 routes. The best description is a power law with an upper
    cutoff, which is typical of networks that grow by preferential attachment: new
    routes go mostly to airports that already have many routes.}
\end{enumerate}

\begin{enumerate}

  \answer{$\alpha = 1.851$, with $x_{\min} = 1.9$ (the lowest average vote). Same MLE

      &n = 4392 && \text{number of movies}\\
      &x_{\min} = \min_i x_i = 1.9 && \text{lowest average vote}\\
      &S = \textstyle\sum_{i=1}^{4392} \ln(x_i / 1.9) = 5163.93 && \text{sum of log ratios}\\

      &\hat\sigma = \sqrt{0.7978} = 0.893 && \text{standard deviation}

  \item \prompt{What is the distribution of data based on your observation? Discuss}
  \answer{%
    \begin{center}
    \begin{tabular}{lcccc}
      \toprule
      & power law & exponential & uniform & normal \\
      \midrule
      KS distance $D$ (data vs model sample) & 0.485 & 0.490 & 0.394 & \textbf{0.055} \\
      \bottomrule
    \end{tabular}
    \end{center}

    \textbf{The movie votes follow a normal distribution, approximately.}
    The data has one peak near the middle of its range and is close to symmetric. The
    normal model predicts 68.3\% of movies within $\pm1\sigma$ and 95.4\% within
    $\pm2\sigma$; the data has 70.1\% and 95.6\%. In the QQ plot the points lie on the
    identity line through the bulk of the data, and the KS distance is about one seventh
    of the next-best model's.

    The other three models fail for structural reasons. The power law and the
    exponential are both monotonically decreasing: their most likely value is the
    smallest one, so they predict mostly terrible movies and a long tail of ratings
    above 10. The uniform model ignores the peak altogether.

    The fit is good but not exact, for two reasons. First, the data is bounded to
    $[1, 10]$ and slightly left-skewed: bad movies can drop further below the average
    than good movies can rise above it. Second, votes are rounded to one decimal, which
    gives the QQ plot its small steps. A normal shape is expected because each value is
    an \emph{average} of many individual votes. By the central limit theorem an average
    of many independent ratings tends toward a normal distribution, whatever the
    distribution of each single rating.}

\prompt{\textbf{Problem 2.}}

\begin{itemize}
  \item \prompt{Screenshot of distribution/histogram of the age}
  \answer{How the ages were computed. The birth dates have the form \texttt{dd-Mon-yy}
    (a few use a 4-digit year). Every 2-digit year is read as 19\textit{yy}: IMDB-WIKI
    was collected in 2015, so a year such as \texttt{20}--\texttt{24} cannot mean
    2020--2024, and the counts per year decrease smoothly from the 1920s down to
    \texttt{00}, with no break that would mark a switch to the 2000s. The 129 rows with
    year \texttt{0000} (a missing-value placeholder) were dropped. Per the professor's
    instruction (2026-09-23), age is now the year difference \textbf{2026 minus birth
    year} (month/day ignored), kept only if $1 \le$ age $\le 100$. Keeping that range
    leaves \textbf{453{,}905} people (mean 56.87, median 55, range 27 to 100). The
    minimum of 27 is a result of the 19\textit{yy} rule, since the latest possible birth
    year is 1999 ($2026-1999=27$). Of the rest, 6{,}684 rows had age $>100$ and were
    dropped; none had age $<1$.}
  \answerfig{p2_1_age}{Age in 2026 ($n = 453{,}905$).}
  \panels{%
    \textbf{Shape.} One broad hump: ages start at 27, rise through the 30s, peak in the 40s and 50s, and fall off in a long right tail up to 100. Most of the mass is between 36 and 71.\par
    \textbf{Jagged bars.} Neighbouring years differ (for example, 48 and 57 stand out). This is year-to-year noise in how many actors were born in a given year, not a pattern in the ages themselves.\par}

  \item \prompt{Screenshot of distribution/histogram of the first digit of age}
  \answerfig{p2_2_first_digit}{First digit of age, with the uniform level $1/9$ (dashed) and Benford's law (orange).}
  \panels{%
    \textbf{Bars.} The shares are 0.2\%, 0.7\%, 7.9\%, 24.5\%, 30.0\%, 18.9\%, 9.9\%, 5.4\% and 2.5\% for digits 1 to 9. They peak at 5 and 4, the same place as the hump in the age histogram.\par
    \textbf{Uniform line.} Far from the dashed $1/9 = 0.111$ level: digits 4 to 6 are well above it, digits 1 to 3 and 8 to 9 well below.\par
    \textbf{Benford curve.} Also far from Benford, which predicts 1 as the most common digit (30.1\%); here 1 is the rarest.\par}

  \item \prompt{Screenshot of distribution/histogram of the last digit of age}
  \answerfig{p2_3_last_digit}{Last digit of age, with the uniform level $1/10$ (dashed).}
  \panels{%
    \textbf{Bars.} Every digit has a share between 8.95\% (digit 3) and 11.10\% (digit 7), within 1.10 percentage points of 10\%.\par
    \textbf{Uniform line.} All ten bars sit close to the dashed 0.100 level, with no digit clearly favoured.\par}

  \item \prompt{Discussion (what are those distribution?) Why do you think it is like this?}
  \answer{%
    \textbf{The last digit is uniform (approximately); the first digit is not.}

    \textbf{Last digit.} This is expected. The last digit of an age is its value modulo
    10. Every block of ten ages (20--29, 30--39, \dots) contains each last digit exactly
    once. The age distribution changes slowly compared with a step of one year, so within
    any block of ten the ten ages have roughly equal counts, and summing over the blocks
    gives each last digit about 10\%. The remaining deviations of up to $\pm1.1$ points
    come from year-to-year noise in how many actors were born in a given year (the jagged
    bars in the age histogram), not from any preference for particular digits.

    \medskip
    \textbf{First digit.} This is also expected, once the two boundary decades are
    accounted for separately. For ages 30 to 99, the first digit \emph{is} the decade of
    life, and grouping by decade reproduces the shape of the age distribution: it peaks
    at 5 (the 50s, 30.0\%) and 4 (the 40s, 24.5\%), matching the median of 55. Digit 2 is
    small (0.7\%) not because few actors are in their 20s, but because the $1 \le
    \text{age} \le 100$ rule only keeps ages 27 to 29 in that decade, three values out of
    ten, a boundary effect of the reference year rather than a property of the data.
    Digit 1 is smaller still (0.2\%): no age from 10 to 19 survives the rule (the
    youngest possible age is 27), so the entire digit-1 count comes from the 816 people
    with age exactly 100 ($816 / 453{,}905 = 0.18\%$, matching the observed share). For
    the first digit to be uniform, every decade would need the same number of actors,
    which working actors clearly are not, and the two extreme digits are further skewed
    by where the $[1, 100]$ window happens to cut.

    \medskip
    \textbf{It is not Benford's law either.} Benford's law (first digit $d$ with
    probability $\log_{10}(1 + 1/d)$, so 1 is the most common at 30.1\%) applies to data
    that spans several orders of magnitude, such as populations, prices or the airport
    route counts in Problem~1, where the log of the value is spread out roughly evenly.
    Age spans less than one order of magnitude (27 to 100) and is concentrated around
    55, so its first digit shows where that one hump sits, not Benford's pattern. In short,
    the last digit is uniform because it throws away where the value is and keeps only
    its position within a block of ten. The first digit keeps exactly that information,
    so its distribution follows the shape of the age distribution.}
\end{itemize}
